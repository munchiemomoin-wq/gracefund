   357→interface ReviewCampaign {
   358→  id: string;
   359→  title: string;
   360→  slug: string;
   361→  goalAmount: number;
   362→  raisedAmount: number;
   363→  status: string;

   365→  organizer: { id: string; name: string | null; avatarUrl: string | null; verificationLevel: string } | null;
   366→  category: { id: string; name: string; slug: string } | null;
   367→  _count: { donations: number; reports: number };
   368→}
   369→
   370→function ReviewQueueView() {
   371→  const [campaigns, setCampaigns] = useState<ReviewCampaign[]>([]);
   372→  const [loading, setLoading] = useState(true);
   373→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   374→
   375→  const fetchCampaigns = useCallback(() => {
   376→    const ac = new AbortController();
   377→    setLoading(true);
   378→    fetch('/api/reviews', { signal: ac.signal })
   379→      .then((r) => (r.ok ? r.json() : []))
   380→      .then((data) => setCampaigns(Array.isArray(data) ? data : []))
   381→      .catch(() => {})
   382→      .finally(() => setLoading(false));
   383→    return () => ac.abort();
   384→  }, []);
   385→
   386→  useEffect(() => { fetchCampaigns(); }, [fetchCampaigns]);
   387→
   388→  const handleAction = async (id: string, action: string) => {
   389→    setActionLoading(id);
   390→    try {
   391→      await fetch(`/api/reviews/${id}`, {
   392→        method: 'PUT',
   393→        headers: { 'Content-Type': 'application/json' },
   394→        body: JSON.stringify({ action }),
   395→      });
   396→      fetchCampaigns();
   397→    } catch {
   398→      /* noop */
   399→    } finally {
   400→      setActionLoading(null);
   401→    }
   402→  };
   403→
   404→  const statusBadge = (status: string) => {
   405→    const map: Record<string, { variant: 'default' | 'secondary' | 'destructive' | 'outline'; label: string }> = {
   406→      submitted: { variant: 'outline', label: 'Submitted' },
   407→      under_review: { variant: 'secondary', label: 'Under Review' },
   408→    };
   409→    const m = map[status] || { variant: 'outline' as const, label: status };
   410→    return <Badge variant={m.variant}>{m.label}</Badge>;
   411→  };
   412→
   413→  return (
   414→    <div className="space-y-6">
   415→      <div className="flex items-center justify-between">
   416→        <div>
   417→          <h2 className="text-lg font-semibold">Campaign Review Queue</h2>
   418→          <p className="text-sm text-muted-foreground">Campaigns awaiting admin review</p>
   419→        </div>
   420→        <Button variant="outline" size="sm" onClick={() => fetchCampaigns()} disabled={loading}>
   421→          <RefreshCw className={`mr-1.5 h-3.5 w-3.5 ${loading ? 'animate-spin' : ''}`} />
   422→          Refresh
   423→        </Button>
   424→      </div>
