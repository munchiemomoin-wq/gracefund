   347→interface ReviewCampaign {
   348→  id: string;
   349→  title: string;
   350→  slug: string;
   351→  goalAmount: number;
   352→  raisedAmount: number;
   353→  status: string;
   354→  submittedAt: string;
   355→  organizer: { id: string; name: string | null; avatarUrl: string | null; verificationLevel: string } | null;
   356→  category: { id: string; name: string; slug: string } | null;
   357→  _count: { donations: number; reports: number };
   358→}
   359→
   360→function ReviewQueueView() {
   361→  const [campaigns, setCampaigns] = useState<ReviewCampaign[]>([]);
   362→  const [loading, setLoading] = useState(true);
   363→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   364→
   365→  const fetchCampaigns = useCallback(() => {
   366→    const ac = new AbortController();
   367→    setLoading(true);
   368→    fetch('/api/reviews', { signal: ac.signal })
   369→      .then((r) => (r.ok ? r.json() : []))
   370→      .then((data) => setCampaigns(Array.isArray(data) ? data : []))
   371→      .catch(() => {})
   372→      .finally(() => setLoading(false));
   373→    return () => ac.abort();
   374→  }, []);
   375→
   376→  useEffect(() => { fetchCampaigns(); }, [fetchCampaigns]);
   377→
   378→  const handleAction = async (id: string, action: string) => {
   379→    setActionLoading(id);
   380→    try {
   381→      await fetch(`/api/reviews/${id}`, {
   382→        method: 'PUT',
   383→        headers: { 'Content-Type': 'application/json' },
   384→        body: JSON.stringify({ action, reviewerId: REVIEWER_ID }),
   385→      });
   386→      fetchCampaigns();
   387→    } catch {
   388→      /* noop */
   389→    } finally {
   390→      setActionLoading(null);
   391→    }
   392→  };
   393→
   394→  const statusBadge = (status: string) => {
   395→    const map: Record<string, { variant: 'default' | 'secondary' | 'destructive' | 'outline'; label: string }> = {
   396→      submitted: { variant: 'outline', label: 'Submitted' },
   397→      under_review: { variant: 'secondary', label: 'Under Review' },
   398→    };
   399→    const m = map[status] || { variant: 'outline' as const, label: status };
   400→    return <Badge variant={m.variant}>{m.label}</Badge>;
   401→  };
   402→
   403→  return (
   404→    <div className="space-y-6">
   405→      <div className="flex items-center justify-between">
   406→        <div>
   407→          <h2 className="text-lg font-semibold">Campaign Review Queue</h2>
   408→          <p className="text-sm text-muted-foreground">Campaigns awaiting admin review</p>
   409→        </div>
   410→        <Button variant="outline" size="sm" onClick={() => fetchCampaigns()} disabled={loading}>
   411→          <RefreshCw className={`mr-1.5 h-3.5 w-3.5 ${loading ? 'animate-spin' : ''}`} />
   412→          Refresh
   413→        </Button>
   414→      </div>
   415→
   416→      {loading ? (
   417→        <ListSkeleton rows={3} />
   418→      ) : campaigns.length === 0 ? (
   419→        <EmptyState icon={ClipboardList} title="No campaigns to review" description="All campaigns have been reviewed. Check back later." />
   420→      ) : (
   421→        <div className="space-y-3">
   422→          {campaigns.map((c) => (
   423→            <Card key={c.id} className="overflow-hidden">
   424→              <CardContent className="p-4 sm:p-5">
   425→                <div className="flex flex-col sm:flex-row sm:items-start justify-between gap-4">

   945→
   946→  const fetchWithdrawals = useCallback((status?: string) => {
   947→    const ac = new AbortController();
   948→    setLoading(true);
   949→    const url = status && status !== 'all' ? `/api/withdrawals?status=${status}` : '/api/withdrawals';
   950→    fetch(url, { signal: ac.signal })
   951→      .then((r) => (r.ok ? r.json() : []))
   952→      .then((data) => setWithdrawals(Array.isArray(data) ? data : []))
   953→      .catch(() => {})
   954→      .finally(() => setLoading(false));

   985→      suspended: { variant: 'destructive', label: 'Suspended' },
   986→    };

   988→    return <Badge variant={v.variant}>{v.label}</Badge>;
   989→  };
   990→
   991→  return (
   992→    <div className="space-y-6">
   993→      <div>

   997→
   998→      <div className="flex flex-wrap gap-2">
   999→        {withdrawalStatusOptions.map((s) => (

  1001→            key={s}
  1002→            size="sm"

  1007→            {s.replace('_', ' ')}

  1009→        ))}

  1011→
  1012→      {loading ? (
  1013→        <ListSkeleton rows={3} />
  1014→      ) : withdrawals.length === 0 ? (
  1015→        <EmptyState icon={Wallet} title="No withdrawal requests" description={filter === 'all' ? 'No withdrawal requests have been made yet.' : `No withdrawals with status "${filter.replace('_', ' ')}".`} />
  1016→      ) : (
  1017→        <ScrollArea className="max-h-96">
  1018→          <div className="space-y-3 pr-4">
  1019→            {withdrawals.map((w) => (
  1020→              <Card key={w.id} className="overflow-hidden">
  1021→                <CardContent className="p-4 sm:p-5">
  1022→                  <div className="flex flex-col sm:flex-row sm:items-start justify-between gap-4">
  1023→                    <div className="min-w-0 flex-1 space-y-1.5">
  1024→                      <div className="flex flex-wrap items-center gap-2">
  1025→                        <h3 className="text-sm font-semibold">{w.campaign?.title || 'Unknown Campaign'}</h3>
  1026→                        {statusBadge(w.status)}
  1027→                      </div>
  1028→                      <div className="flex flex-wrap items-center gap-3 text-xs text-muted-foreground">
  1029→                        <span className="font-medium text-foreground">{formatCurrency(w.amount)}</span>
  1030→                        {w.requester && <span>by {w.requester.name}</span>}
  1031→                        <span>Requested {new Date(w.requestedAt).toLocaleDateString()}</span>
  1032→                      </div>
  1033→                      {w.campaign && (
  1034→                        <p className="text-xs text-muted-foreground">
  1035→                          Campaign: {formatCurrency(w.campaign.raisedAmount)} raised of {formatCurrency(w.campaign.goalAmount)} goal
  1036→                        </p>
  1037→                      )}
  1038→                      {w.rejectionReason && (
  1039→                        <p className="text-xs text-red-500">Rejected: {w.rejectionReason}</p>
  1040→                      )}
  1041→                      {w.reviewNotes && (
  1042→                        <p className="text-xs text-muted-foreground">Notes: {w.reviewNotes}</p>
  1043→                      )}
  1044→                    </div>
  1045→                    {(w.status === 'requested' || w.status === 'under_review') && (
  1046→                      <div className="flex flex-col gap-2 shrink-0">
  1047→                        <Button
  1048→                          size="sm"
  1049→                          variant="outline"
  1050→                          className="h-8 text-xs text-red-600 hover:text-red-700 hover:bg-red-50 dark:hover:bg-red-950/30"

  1060→                          onClick={() => handleAction(w.id, 'approved')}
  1061→                        >
  1062→                          {updatingId === w.id ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : 'Approve'}
  1063→                        </Button>
  1064→                      </div>
  1065→                    )}
  1066→                    {w.status === 'approved' && (
  1067→                      <Button
  1068→                        size="sm"
  1069→                        className="h-8 text-xs bg-sky-600 hover:bg-sky-700"

  1071→                        onClick={() => handleAction(w.id, 'paid')}
  1072→                      >
  1073→                        {updatingId === w.id ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : 'Mark Paid'}
  1074→                      </Button>
  1075→                    )}
  1076→                  </div>
  1077→                </CardContent>
  1078→              </Card>
  1079→            ))}
  1080→          </div>
  1081→        </ScrollArea>
  1082→      )}
  1083→    </div>
  1084→  );
